# Dois pilotos h300 com 16 threads

Este bloco usa somente o engine c493be8c já qualificado e merged.
Verifica os dois pacotes anteriores e cria apenas os dois claims ainda não consumidos.
Se um start falhar, preserve os recibos e envie a saída; não repita o bloco.
Ele não submete o grafo futuro de 95 jobs.

```bash
(
set -euo pipefail
BASE=/home/vrrcelestino/model-agrologistic-hygiene-audit
PY=/home/vrrcelestino/venv313/bin/python
SOURCE_SHA=c493be8cd127c5175fc7eb111400229e64b8e3c0
DRIVER="$BASE/.mvp2-pyomo-scaling-public-v1/source/scripts/pyomo_scaling_campaign.py"
CAMPAIGN="$BASE/.mvp2-pyomo-scaling-native8h-v1"
export GRB_LICENSE_FILE=/home/vrrcelestino/model-agrologistic/secrets/gurobi.lic
test -x "$PY"
test -d "$CAMPAIGN"
printf '%s  %s\n' \
  '27d90f396d50a6778ca7ccb0bf2f11878186ef58e343128aabedcb1057b4ec2d' \
  "$DRIVER" | sha256sum -c -

for CELL in h300-w-sto-highs-t16-r1 h300-w-sto-gurobi-t16-r1
do
  if [[ -e "$CAMPAIGN/$CELL" || -L "$CAMPAIGN/$CELL" ]]; then
    printf 'PRESERVE_EXISTING_CLAIM=%s; envie o recibo, sem repetir start\n' "$CAMPAIGN/$CELL"
    exit 2
  fi
done

"$PY" -B -I - "$CAMPAIGN" <<'PY'
import hashlib
import json
import pathlib
import sys
base = pathlib.Path(sys.argv[1])
expected = (
    ('gurobi', '2231228', 'a07f2629dbdd663dbc0a5549d4cdc5f026c866507a884c1c8cb53c9e277985f1'),
    ('highs', '2231226', '412f4d4d3b7faa7cb913a4da3a537a6804926840fb859bfc22bfd72e11fe702b'),
)
checked = []
for solver, job, checksum in expected:
    claim = base / f'h300-w-sto-{solver}-t1-r1'
    assert claim.is_dir() and not claim.is_symlink()
    marker = claim / 'collection.json'
    archive = claim / 'evidence.tar.gz'
    assert marker.is_file() and not marker.is_symlink()
    assert archive.is_file() and not archive.is_symlink()
    record = json.loads(marker.read_bytes())
    assert record['sha256'] == checksum
    with archive.open('rb') as stream:
        assert hashlib.file_digest(stream, 'sha256').hexdigest() == checksum
    row = record['observation']
    assert row['job_id'] == job
    assert row['source_commit'] == 'c493be8cd127c5175fc7eb111400229e64b8e3c0'
    assert row['campaign_sha256'] == '3cae7c318032108caa936d70b2883f20d055066f89a78e24610f261832763b58'
    assert row['cell']['threads'] == 1 and row['cell']['seed'] == 42
    assert row['cell']['solver'] == solver and row['cell']['population'] == 300
    assert row['operational_receipts_closed'] is True
    assert row['native_result_returned'] is True
    assert row['incumbent_validation_rejected'] is False
    resource = row['resource_observation']
    assert resource['status'] == 'timely_samples'
    assert 0 < resource['sampled_worker_tree_peak_bytes'] <= 0.8 * 192 * 1024**3
    checked.append({'solver': solver, 'job_id': job, 'original_archive_preserved': True})
print(json.dumps({'status': 'two_original_t1_anchors_verified', 'observations': checked}))
PY

for CELL in h300-w-sto-highs-t16-r1 h300-w-sto-gurobi-t16-r1
do
  printf '\nNOVO PILOTO: %s\n' "$CELL"
  "$PY" -B -I "$DRIVER" start \
    --directory "$CAMPAIGN" --source-sha "$SOURCE_SHA" \
    --cell "$CELL" --acknowledge-one-cell
done
printf '\nTWO_H300_T16_PILOTS_ONLY; ORIGINAL_T1_CLAIMS_PRESERVED; NO_AUTOMATIC_FOLLOW_ON\n'
)
```
